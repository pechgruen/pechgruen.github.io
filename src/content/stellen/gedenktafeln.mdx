<script>{`
  (function () {
    const img = document.getElementById('tafel-alle');
    const map = document.getElementById('tafeln');
    if (!img || !map) return;

    // --- responsive coords ---
    const origW = parseFloat(img.dataset.w || '0');
    const origH = parseFloat(img.dataset.h || '0');
    const areas = Array.from(map.querySelectorAll('area[data-coords]'));

    function rescale() {
      const rect = img.getBoundingClientRect();
      if (!rect.width || !rect.height) return;
      const scaleX = rect.width / origW;
      const scaleY = rect.height / origH;

      areas.forEach(area => {
        const raw = (area.dataset.coords || '')
          .split(',')
          .map(s => parseFloat(s.trim()))
          .filter(n => Number.isFinite(n));
        if (raw.length < 6) return;

        const scaled = raw.map((v, i) => {
          const s = (i % 2 === 0) ? scaleX : scaleY;
          return Math.round(v * s);
        });
        area.coords = scaled.join(',');
      });
    }

    if (img.complete) rescale();
    img.addEventListener('load', rescale);
    window.addEventListener('resize', rescale, { passive: true });
    setTimeout(rescale, 50);
    setTimeout(rescale, 250);

    // --- lightbox ---
    const lb = document.getElementById('tafel-lightbox');
    const lbImg = document.getElementById('tafel-lightbox-img');
    const lbClose = document.getElementById('tafel-lightbox-close');

    let lastFocus = null;

    function openLightbox(src, alt) {
      if (!lb || !lbImg) return;
      lastFocus = document.activeElement;

      lbImg.src = src;
      lbImg.alt = alt || '';

      lb.classList.add('is-open');
      lb.setAttribute('aria-hidden', 'false');
      document.body.style.overflow = 'hidden';

      // Fokus auf Close-Button (besser für Keyboard)
      if (lbClose) lbClose.focus();
    }

    function closeLightbox() {
      if (!lb) return;
      lb.classList.remove('is-open');
      lb.setAttribute('aria-hidden', 'true');
      document.body.style.overflow = '';

      // Bild leeren (optional)
      if (lbImg) {
        lbImg.src = '';
        lbImg.alt = '';
      }

      if (lastFocus && typeof lastFocus.focus === 'function') lastFocus.focus();
    }

    // Klick auf Hotspot -> Overlay statt Navigation
    map.addEventListener('click', (e) => {
      const t = e.target;
      if (!t || t.tagName !== 'AREA') return;

      const src = t.getAttribute('data-full') || t.getAttribute('href');
      const alt = t.getAttribute('alt') || '';

      if (src) {
        e.preventDefault();
        openLightbox(src, alt);
      }
    });

    // Klick auf Hintergrund schließt
    lb?.addEventListener('click', (e) => {
      if (e.target === lb) closeLightbox();
    });

    lbClose?.addEventListener('click', closeLightbox);

    // ESC schließt
    document.addEventListener('keydown', (e) => {
      if (e.key === 'Escape' && lb?.classList.contains('is-open')) {
        closeLightbox();
      }
    });
  })();
`}</script>
